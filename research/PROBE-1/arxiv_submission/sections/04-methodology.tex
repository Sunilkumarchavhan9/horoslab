\section{Proposed Methodology}
\label{sec:methodology}

\subsection{Adaptive Conformal Friction Calibration}
Standard tactile friction estimators produce point estimates $\hat{\mu}_i(k)$. Under abrupt liquid spills or surface transitions, empirical residuals violate exchangeability. To supply certified lower bounds without ad-hoc safety heuristics, we integrate Adaptive Conformal Inference (ACI) \cite{gibbs2021adaptive}.

Let $\alpha \in (0, 1)$ denote the target miscoverage rate (e.g., $\alpha = 0.05$ for $95\%$ coverage). At each control step $k$:
\begin{enumerate}
    \item \textbf{Non-Conformity Scoring:} We evaluate the absolute prediction residual:
    \begin{equation}
        E_k = |\mu_{\text{observed}, i}(k) - \hat{\mu}_i(k)|
    \end{equation}
    \item \textbf{Online Quantile Tracking:} The empirical miscoverage parameter $\alpha_t$ updates via a stochastic approximation scheme:
    \begin{equation}
        \alpha_{t+1} = \alpha_t + \beta \left( \alpha - \mathbb{I}\{E_t > Q_t\} \right)
        \label{eq:aci_update}
    \end{equation}
    where $\beta = 0.05$ is the adaptation step size, $\mathbb{I}\{\cdot\}$ is the indicator function, and $Q_t = \text{Quantile}_{1 - \alpha_t}(E_{1:t-1})$.
    \item \textbf{Calibrated Lower Bound:} The friction coefficient passed to the cone constraints \eqref{eq:cone_rows} is:
    \begin{equation}
        \mu_{\text{cal}, i}(k) = \max\left( \mu_{\text{floor}}, \hat{\mu}_i(k) - Q_{1 - \alpha_{t+1}}(E_{1:t}) \right)
        \label{eq:mu_cal}
    \end{equation}
    where $\mu_{\text{floor}} = 0.05$ guarantees physical non-negativity.
\end{enumerate}

\begin{proposition}[Long-Run Average Coverage \cite{gibbs2021adaptive}]
\label{prop:aci_coverage}
Under the ACI update rule \eqref{eq:aci_update} with bounded residuals $E_t \in [0, B]$, the asymptotic empirical miscoverage rate satisfies:
\begin{equation}
    \lim_{T \to \infty} \frac{1}{T} \sum_{t=1}^T \mathbb{I}\{E_t > Q_t\} = \alpha + \mathcal{O}\left( \frac{1}{\beta T} \right)
\end{equation}
irrespective of the nonstationary sequence $\{\mu_i(k)\}_{k=1}^T$.
\end{proposition}

\subsection{Quadratic Programming Predictive Safety Filter}
The safe joint acceleration command $\mathbf{u}^* \in \mathbb{R}^{n_q}$ is solved at each 100~Hz control cycle via the following convex QP:
\begin{equation}
    \min_{\mathbf{u}, \boldsymbol{\delta}_s, \boldsymbol{\delta}_w} \frac{1}{2} \|\mathbf{u} - \mathbf{u}_{\text{nom}}\|_{\mathbf{W}}^2 + \frac{1}{2} \rho_s \|\boldsymbol{\delta}_s\|_2^2 + \frac{1}{2} \rho_w \|\boldsymbol{\delta}_w\|_2^2
    \label{eq:safety_qp}
\end{equation}
\textbf{Subject to:}
\begin{align}
    \mathbf{A}_{\text{ineq}} \mathbf{u} - \boldsymbol{\delta}_s &\le \mathbf{b}_{\text{ineq}} \label{eq:qp_ineq} \\
    \mathbf{A}_{\text{eq}} \mathbf{u} - \boldsymbol{\delta}_w &= \mathbf{b}_{\text{eq}} \label{eq:qp_eq} \\
    \boldsymbol{\delta}_s \ge \mathbf{0}, \quad \mathbf{u}_{\min} &\le \mathbf{u} \le \mathbf{u}_{\max} \label{eq:qp_bounds}
\end{align}

Here, $\mathbf{u}_{\text{nom}}$ is the nominal unconstrained action command dispatched by an upstream manipulation policy (e.g., flow matching, diffusion, or teleoperation), and $\mathbf{W} = \mathbf{I}_{n_q}$. Soft slack variables $\boldsymbol{\delta}_s \in \mathbb{R}^{8M + 2M + 2n_q}$ and $\boldsymbol{\delta}_w \in \mathbb{R}^6$ ensure continuous recursive feasibility. Penalty weights are configured as $\rho_s = 10^5$ and $\rho_w = 10.0$, strictly penalizing barrier violations while allowing slight compliant adaptation in net wrench balance.

The constraint matrices aggregate the multi-contact dynamics:
\begin{equation}
    \mathbf{A}_{\text{ineq}} = \begin{bmatrix}
        \mathbf{A}_{\text{cone}, 1} \mathbf{\Lambda}_1 \\
        \vdots \\
        \mathbf{A}_{\text{cone}, M} \mathbf{\Lambda}_M \\
        \hat{\mathbf{n}}_1^T \mathbf{\Lambda}_1 \\
        \vdots \\
        -\hat{\mathbf{n}}_M^T \mathbf{\Lambda}_M \\
        \mathbf{I}_{n_q} \\
        -\mathbf{I}_{n_q}
    \end{bmatrix}, \quad
    \mathbf{b}_{\text{ineq}} = \begin{bmatrix}
        (1 - \gamma_{\text{slip}}) \mathbf{p}_{\text{curr}, 1} - \mathbf{A}_{\text{cone}, 1} \mathbf{f}_{c, 1}^{\text{bias}} \\
        \vdots \\
        (1 - \gamma_{\text{slip}}) \mathbf{p}_{\text{curr}, M} - \mathbf{A}_{\text{cone}, M} \mathbf{f}_{c, M}^{\text{bias}} \\
        F_{\max} - \hat{\mathbf{n}}_1^T \mathbf{f}_{c, 1}^{\text{bias}} \\
        \vdots \\
        -F_{\min} + \hat{\mathbf{n}}_M^T \mathbf{f}_{c, M}^{\text{bias}} \\
        \mathbf{u}_{\max} \\
        -\mathbf{u}_{\min}
    \end{bmatrix}
\end{equation}
and $\mathbf{A}_{\text{eq}} = \mathbf{G} \mathbf{\Lambda}_{\text{stack}}$, $\mathbf{b}_{\text{eq}} = \mathbf{w}_{\text{req}} - \mathbf{G} \mathbf{f}_{\text{bias, stack}}$.

\subsection{Four-State Decision Gate and Passive Fallback}
To ensure deterministic execution in physical real-time systems, the output of the QP is governed by a finite-state machine:
\begin{enumerate}
    \item \textbf{\texttt{NOMINAL\_SAFE}:} The solver completes within the $10.0\text{ ms}$ budget with maximum slack $\|\boldsymbol{\delta}_s\|_\infty \le 10^{-2}\text{ N}$. Optimal command $\mathbf{u}^*$ is dispatched directly to actuator drivers.
    \item \textbf{\texttt{EMERGENCY\_RELAXED}:} When severe external perturbations force $\|\boldsymbol{\delta}_s\|_\infty > 10^{-2}\text{ N}$, the controller commands the relaxed solution while incrementing a safety counter. If relaxed execution persists for $> 5$ consecutive cycles ($50\text{ ms}$), the system trips to \texttt{CRITICAL\_FAIL}.
    \item \textbf{\texttt{TIMEOUT}:} If the solver runtime exceeds the $10.0\text{ ms}$ watchdog deadline, execution immediately aborts to \texttt{CRITICAL\_FAIL}.
    \item \textbf{\texttt{CRITICAL\_FAIL}:} The system dispatches passive damping:
    \begin{equation}
        \mathbf{u}_{\text{safe}} = -K_d \dot{\mathbf{q}}_k
    \end{equation}
    where $K_d = 5.0\text{ N}\cdot\text{s/rad}$. This command executes in $< 50\ \mu\text{s}$ without optimization, absorbing kinetic energy and stabilizing the fingers against the object.
\end{enumerate}
